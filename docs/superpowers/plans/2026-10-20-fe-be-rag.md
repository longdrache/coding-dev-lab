# FE → BE → RAG Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Cho học viên hỏi trợ lý AI trong trang bài tập và cho admin nhập tài liệu, với mọi lời gọi bot RAG đi qua BE.

**Architecture:** BE thêm module `rag` gồm `RagBotClient` (giữ `BOT_URL`/`ADMIN_KEY`, fetch + timeout + map lỗi), `RagService` (validate + trích `answer`) và `RagController` (`POST /api/rag/chat`, cần đăng nhập). Admin ingest nằm trong `AdminModule` tại `POST /api/admin/rag/ingest`. Admin gọi BE qua BFF catch-all hiện có; FE gọi thẳng BE với cookie `session`.

**Tech Stack:** NestJS 12 (ESM, `.ts` import), vitest (BE + FE), Next.js 16 App Router (FE + admin), `jose` không còn cần cho route RAG của admin.

**Spec:** `docs/superpowers/specs/2026-10-20-fe-be-rag-design.md`

## Global Constraints

- Mọi import nội bộ trong `be/` phải có đuôi `.ts` (ESM), ví dụ `import { RagService } from './rag.service.ts';`.
- `BOT_URL` và `ADMIN_KEY` chỉ đọc ở BE; tuyệt đối không dùng tiền tố `NEXT_PUBLIC`.
- Không log nội dung `message`/`history`/`text`.
- Copy hướng tới người dùng bằng tiếng Việt, không emoji.
- BE test bằng vitest, mock `fetch` bằng `vi.stubGlobal('fetch', …)`; FE test bằng vitest môi trường `node` (chỉ file `*.test.ts`); admin test bằng `node --test`.

- Commit chỉ thực hiện khi người dùng yêu cầu; các bước commit dưới đây là mẫu.

```ts
// be/src/rag/rag.bot-client.ts
import { BadGatewayException, Injectable, ServiceUnavailableException } from '@nestjs/common';

export type BotResponse = { status: number; ok: true; text: string };

@Injectable()
export class RagBotClient {
  private config(): { baseUrl: string; adminKey: string } {
    const baseUrl = (process.env.BOT_URL ?? '').trim().replace(/\/+$/, '');
    const adminKey = (process.env.ADMIN_KEY ?? '').trim();
    if (!baseUrl || !adminKey) {
      throw new ServiceUnavailableException('Chưa cấu hình BOT_URL/ADMIN_KEY');
    }
    return { baseUrl, adminKey };
  }

  async post(path: string, body: unknown, timeoutMs: number): Promise<BotResponse> {
    const { baseUrl, adminKey } = this.config();

    let response: Response;
    try {
      response = await fetch(`${baseUrl}${path}`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          Authorization: `Bearer ${adminKey}`,
        },
        body: JSON.stringify(body),
        signal: AbortSignal.timeout(timeoutMs),
      });
    } catch {
      throw new BadGatewayException('Không kết nối được bot RAG');
    }

    const text = await response.text();
    if (!response.ok) {
      throw new BadGatewayException(`Bot RAG trả lỗi ${response.status}`);
    }
    return { status: response.status, ok: true, text };
  }
}
```

```ts
// be/src/rag/rag.service.ts
import { BadGatewayException, BadRequestException, Injectable } from '@nestjs/common';
import { RagBotClient } from './rag.bot-client.ts';

export type ChatRole = 'user' | 'assistant';
export type ChatTurn = { role: ChatRole; content: string };
export type ChatInput = { message: string; history: ChatTurn[]; problemSlug?: string };

export const MAX_MESSAGE_LENGTH = 2000;
export const MAX_HISTORY_TURNS = 10;
const SLUG_RE = /^[a-z0-9-]{1,120}$/;

export function normalizeChatInput(raw: Record<string, unknown>): ChatInput {
  const message = typeof raw.message === 'string' ? raw.message.trim() : '';
  if (!message) throw new BadRequestException('Thiếu câu hỏi');
  if (message.length > MAX_MESSAGE_LENGTH) {
    throw new BadRequestException(`Câu hỏi tối đa ${MAX_MESSAGE_LENGTH} ký tự`);
  }

  const history = (
    Array.isArray(raw.history)
      ? raw.history
          .map((turn): ChatTurn | null => {
            if (!turn || typeof turn !== 'object') return null;
            const { role, content } = turn as { role?: unknown; content?: unknown };
            if (role !== 'user' && role !== 'assistant') return null;
            if (typeof content !== 'string') return null;
            const trimmed = content.trim();
            if (!trimmed) return null;
            return { role, content: trimmed.slice(0, MAX_MESSAGE_LENGTH) };
          })
          .filter((turn): turn is ChatTurn => turn !== null)
      : []
  ).slice(-MAX_HISTORY_TURNS);

  const rawSlug = typeof raw.problemSlug === 'string' ? raw.problemSlug.trim() : '';
  const problemSlug = rawSlug && SLUG_RE.test(rawSlug) ? rawSlug : undefined;

  return { message, history, problemSlug };
}

export function extractAnswer(text: string): string {
  const trimmed = text.trim();
  if (!trimmed) return '';
  try {
    const data = JSON.parse(trimmed) as Record<string, unknown>;
    for (const key of ['answer', 'reply', 'message', 'response'] as const) {
      const value = data?.[key];
      if (typeof value === 'string' && value.trim()) return value.trim();
    }
  } catch {
    // không phải JSON: dùng text thô
  }
  return trimmed;
}

@Injectable()
export class RagService {
  constructor(private readonly bot: RagBotClient) {}

  async ask(raw: unknown): Promise<{ answer: string }> {
    const input = normalizeChatInput((raw ?? {}) as Record<string, unknown>);
    const res = await this.bot.post(
      '/chat',
      {
        message: input.message,
        history: input.history,
        context: { problemSlug: input.problemSlug },
      },
      30_000,
    );
    const answer = extractAnswer(res.text);
    if (!answer) throw new BadGatewayException('Bot RAG trả về nội dung rỗng');
    return { answer };
  }
}
```

```ts
// be/src/rag/rag.controller.ts
import { Body, Controller, Post, UseGuards } from '@nestjs/common';
import {
  ApiBearerAuth,
  ApiCookieAuth,
  ApiOperation,
  ApiResponse,
  ApiTags,
} from '@nestjs/swagger';
import { AuthGuard } from '../auth/auth.guard.ts';
import { Throttle } from '../common/throttle.guard.ts';
import { RagService } from './rag.service.ts';

@ApiTags('rag')
@Controller('api/rag')
export class RagController {
  constructor(private readonly rag: RagService) {}

  @Post('chat')
  @ApiBearerAuth()
  @ApiCookieAuth()
  @ApiOperation({ summary: 'Hỏi trợ lý AI (cần đăng nhập, 10 req/phút)' })
  @ApiResponse({ status: 201, description: '`{ answer }`.' })
  @ApiResponse({ status: 401, description: 'Thiếu/sai phiên đăng nhập.' })
  @ApiResponse({ status: 429, description: 'Quá 10 req/phút.' })
  @ApiResponse({ status: 502, description: 'Bot RAG lỗi hoặc không kết nối được.' })
  @UseGuards(AuthGuard)
  @Throttle({ default: { limit: 10, ttl: 60_000 } })
  chat(@Body() body: unknown) {
    return this.rag.ask(body);
  }
}
```

```ts
// be/src/admin/admin-rag.controller.ts
import { BadRequestException, Body, Controller, Post, UseGuards } from '@nestjs/common';
import { AdminGuard } from './admin.guard.ts';
import { RagBotClient } from '../rag/rag.bot-client.ts';

export const MAX_INGEST_BYTES = 5 * 1024 * 1024;
export const MAX_SOURCE_LENGTH = 120;

@Controller('api/admin/rag')
@UseGuards(AdminGuard)
export class AdminRagController {
  constructor(private readonly bot: RagBotClient) {}

  @Post('ingest')
  async ingest(@Body() body: { source?: unknown; text?: unknown }) {
    const source = typeof body?.source === 'string' ? body.source.trim() : '';
    const text = typeof body?.text === 'string' ? body.text : '';

    if (!source) throw new BadRequestException("Thiếu 'source'");
    if (source.length > MAX_SOURCE_LENGTH) {
      throw new BadRequestException(`'source' tối đa ${MAX_SOURCE_LENGTH} ký tự`);
    }
    if (!text.trim()) throw new BadRequestException('Tài liệu rỗng');
    if (Buffer.byteLength(text, 'utf8') > MAX_INGEST_BYTES) {
      throw new BadRequestException('Tài liệu tối đa 5MB');
    }

    await this.bot.post('/ingest', { source, text }, 60_000);
    return { ok: true };
  }
}
```

```ts
// FE/app/problem/assistant.ts
export type ChatRole = "user" | "assistant";
export type ChatTurn = { role: ChatRole; content: string };

export const MAX_HISTORY_TURNS = 10;

export function withTurn(history: ChatTurn[], turn: ChatTurn): ChatTurn[] {
  return [...history, turn].slice(-MAX_HISTORY_TURNS);
}
```

- Modify: `FE/app/problem/[slug]/page.tsx` (import + render sau khối tab, trước `</section>`)

**Interfaces:**
- Consumes: `useSession()` (`FE/app/ui/AuthProvider.tsx`), `withTurn` + `ChatTurn` (Task 6), `POST /api/rag/chat`.
- Produces: `<ProblemAssistant slug={problem.slug} />`.

- [ ] **Step 1: Tạo component**

```tsx
// FE/components/ProblemAssistant.tsx
"use client";

import { useEffect, useRef, useState } from "react";
import { Bot, Loader2, Send } from "lucide-react";
import { useSession } from "@/app/ui/AuthProvider";
import { withTurn, type ChatTurn } from "@/app/problem/assistant";

const API_URL = process.env.NEXT_PUBLIC_API_URL ?? "http://localhost:4000";

export function ProblemAssistant({ slug }: { slug: string }) {
  const { user, loading } = useSession();
  const [open, setOpen] = useState(false);
  const [history, setHistory] = useState<ChatTurn[]>([]);
  const [input, setInput] = useState("");
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const scrollRef = useRef<HTMLDivElement | null>(null);

  useEffect(() => {
    scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight });
  }, [history, busy]);

  if (loading) return null;

  if (!user) {
    return (
      <div className="mt-5 rounded-xl border border-dashed border-zinc-300 px-4 py-3 text-sm text-zinc-500">
        Đăng nhập để hỏi trợ lý AI về bài này.
      </div>
    );
  }

  async function send() {
    const message = input.trim();
    if (!message || busy) return;

    const previous = history;
    const nextHistory = withTurn(history, { role: "user", content: message });

    setError(null);
    setInput("");
    setHistory(nextHistory);
    setBusy(true);
    try {
      const res = await fetch(`${API_URL}/api/rag/chat`, {
        method: "POST",
        credentials: "include",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ message, history: nextHistory, problemSlug: slug }),
      });
      const data = (await res.json().catch(() => null)) as {
        answer?: string;
        message?: string;
      } | null;

      if (!res.ok || !data?.answer) {
        setHistory(previous);
        if (res.status === 401) setError("Phiên đã hết hạn, hãy đăng nhập lại.");
        else if (res.status === 429) setError("Bạn hỏi hơi nhanh, thử lại sau một phút.");
        else setError(data?.message ?? "Trợ lý đang bận, thử lại sau.");
        return;
      }

      setHistory((prev) => withTurn(prev, { role: "assistant", content: data.answer as string }));
    } catch {
      setHistory(previous);
      setError("Không kết nối được trợ lý.");
    } finally {
      setBusy(false);
    }
  }

  return (
    <div className="mt-5 rounded-xl border border-zinc-200 bg-zinc-50/60">
      <button
        type="button"
        onClick={() => setOpen((value) => !value)}
        className="flex w-full items-center justify-between px-4 py-3 text-left"
      >
        <span className="flex items-center gap-2 text-sm font-semibold text-zinc-900">
          <Bot className="size-4 text-emerald-600" /> Trợ lý AI
        </span>
        <span className="text-xs text-zinc-500">
          {open ? "Thu gọn" : "Hỏi về bài này"}
        </span>
      </button>

      {open && (
        <div className="border-t border-zinc-200 p-3">
          <div ref={scrollRef} className="max-h-64 space-y-2 overflow-y-auto pr-1">
            {history.length === 0 && (
              <p className="text-xs text-zinc-500">
                Hỏi về hướng làm, độ phức tạp, hoặc lỗi bạn đang gặp.
              </p>
            )}
            {history.map((turn, index) => (
              <div key={index} className={turn.role === "user" ? "text-right" : ""}>
                <p
                  className={`inline-block max-w-[85%] whitespace-pre-wrap rounded-xl px-3 py-2 text-sm ${
                    turn.role === "user"
                      ? "bg-zinc-900 text-white"
                      : "border border-zinc-200 bg-white text-zinc-800"
                  }`}
                >
                  {turn.content}
                </p>
              </div>
            ))}
            {busy && (
              <p className="flex items-center gap-2 text-xs text-zinc-500">
                <Loader2 className="size-3.5 animate-spin" /> Đang trả lời...
              </p>
            )}
          </div>

          {error && <p className="mt-2 text-xs text-rose-600">{error}</p>}

          <div className="mt-2 flex gap-2">
            <textarea
              value={input}
              onChange={(event) => setInput(event.target.value)}
              onKeyDown={(event) => {
                if (event.key === "Enter" && !event.shiftKey) {
                  event.preventDefault();
                  void send();
                }
              }}
              rows={2}
              maxLength={2000}
              placeholder="Nhập câu hỏi..."
              className="min-h-[42px] flex-1 resize-none rounded-lg border border-zinc-300 px-3 py-2 text-sm outline-none focus:border-emerald-500"
            />
            <button
              type="button"
              onClick={() => void send()}
              disabled={busy || !input.trim()}
              aria-label="Gửi câu hỏi"
              className="self-end rounded-lg bg-emerald-600 px-3 py-2 text-white transition hover:bg-emerald-700 disabled:opacity-50"
            >
              <Send className="size-4" />
            </button>
          </div>
        </div>
      )}
    </div>
  );
}
```
